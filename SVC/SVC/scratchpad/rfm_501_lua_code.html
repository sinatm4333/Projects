267function getQuery_page(queryType , pageFrom , perPage , pageTo)
268 local slicePageNumber = "";

270 local limit = tonumber(perPage);

274 local limit = tonumber(perPage)*(tonumber(pageTo) - tonumber(pageFrom) + 1) ;

281function getQuery_result(queryType , dataQuery)
282 local resultExp = nil;
283 
284 db.query(dataQuery)
285 local record={};
286 if queryType == _QUERY_TYPE._TYPE_TOTAL_COUNT then
287 record = db.query_fetch();
288 resultExp = record[1];
289 elseif queryType == _QUERY_TYPE._TYPE_TOTAL_SUM then
290 local columnsSelected = {};

292 for i = 1, #columns , 1 do
293 local itemColumns = columns[i];
294 if itemColumns.key ~= nil and itemColumns.sum ~= nil and itemColumns.sum == true then
295 table.insert(columnsSelected , itemColumns.key)
296 end
297 end
298 if #columns > 0 then
299 resultExp = {};
300 record = db.query_fetch();
301 for i = 1, #columnsSelected , 1 do
302 resultExp[columnsSelected[i]] = record[i];
303 end
304 end
305 else
306 resultExp = {};
307 while db.query_fetch(record) do
308 local itemRow = {};
309 local columns = getTableConfig();
310 for i = 1, #columns , 1 do
311 local itemColumns = columns[i];
312 if itemColumns.key ~= nil then
313 itemRow[itemColumns.key] = record[i];
314 end
315 end
316 table.insert(resultExp, itemRow)
317 end
318 end
319 db.query_free();
